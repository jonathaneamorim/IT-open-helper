sudo dnf install postgres

sudo postgresql-setup --initdb -> Inicializar o cluster de dados
sudo chown -R postgres:postgres /var/lib/pgsql/data -> Ajustar permissões (geralmente automático)

sudo systemctl status postgresql -> Verificar status
sudo systemctl start postgresql -> Iniciar
sudo systemctl enable postgresql -> inicialize no boot

sudo -i -u postgres
psql
